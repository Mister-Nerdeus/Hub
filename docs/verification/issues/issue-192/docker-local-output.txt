    environment:
      POSTGRES_DB: hub
      POSTGRES_PASSWORD: hub
      POSTGRES_USER: hub

        - pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}
      timeout: 5s
      interval: 5s
      retries: 12
    image: postgres:16-alpine
    networks:
      default: null
    volumes:
      - type: volume
        source: hub-postgres-data
        target: /var/lib/postgresql/data

      VITE_API_BASE_URL: http://localhost:8010
    networks:
      default: null
    ports:
      - mode: ingress
        target: 5173
        published: "5180"
        protocol: tcp
