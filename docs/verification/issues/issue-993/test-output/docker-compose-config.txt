name: hub
services:
  api:
    build:
      context: C:\Projects\Hub
      dockerfile: apps/api/Dockerfile.local
    depends_on:
      db:
        condition: service_healthy
        required: true
    environment:
      API_HOST: 0.0.0.0
      API_PORT: "8000"
      APP_ENV: local
      CORS_ORIGINS: http://localhost:5180,http://localhost:5173,http://localhost:5174
      DATABASE_URL: postgresql+psycopg://hub:hub@db:5432/hub
    networks:
      default: null
    ports:
      - mode: ingress
        target: 8000
        published: "8010"
        protocol: tcp
  db:
    environment:
      POSTGRES_DB: hub
      POSTGRES_PASSWORD: hub
      POSTGRES_USER: hub
    healthcheck:
      test:
        - CMD-SHELL
        - pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}
      timeout: 5s
      interval: 5s
      retries: 12
    image: postgres:16-alpine
    networks:
      default: null
    volumes:
      - type: volume
        source: hub-postgres-data
        target: /var/lib/postgresql/data
        volume: {}
  web:
    build:
      context: C:\Projects\Hub
      dockerfile: apps/web/Dockerfile.local
    depends_on:
      api:
        condition: service_started
        required: true
    environment:
      VITE_API_BASE_URL: ""
      VITE_API_PROXY_TARGET: http://api:8000
      VITE_BUILD_COMMIT: local-dev
      VITE_BUILD_TIME: local-dev
    healthcheck:
      test:
        - CMD
        - node
        - -e
        - fetch('http://127.0.0.1:5173/?section=editor').then((response) => { if (!response.ok) process.exit(1); }).catch(() => process.exit(1));
      timeout: 4s
      interval: 12s
      retries: 12
    networks:
      default: null
    ports:
      - mode: ingress
        target: 5173
        published: "5180"
        protocol: tcp
networks:
  default:
    name: hub_default
volumes:
  hub-postgres-data:
    name: hub_hub-postgres-data
